## Store
```ts
import { openDB } from 'idb';
export const db = openDB('todos', 1, { upgrade(db) { db.createObjectStore('todos', { keyPath: 'id' }); } });
export async function toggleTodo(id: string) {
  const store = (await db).transaction('todos', 'readwrite').store;
  const todo = await store.get(id);
  await store.put({ ...todo, done: !todo.done });
}
```
